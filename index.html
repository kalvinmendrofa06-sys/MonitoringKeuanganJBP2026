const SPREADSHEET_ID = "ISI_DENGAN_ID_SPREADSHEET"; const NAMA_SHEET = "Data Pengeluaran"; const FOLDER_ID = "1RWxQZQolp14z7qH-8k5wEaLasLgwnonq"; const HEADER_SHEET = [ "ID Transaksi", "Timestamp", "Nama Pengaju", "Divisi", "Tanggal Transaksi", "No. Nota", "Bukti Nota", "Jumlah Barang", "Nama Barang", "Harga per Satuan", "Total", "Client ID" ]; // ===================================================== // GET // ===================================================== function doGet(e) { // Cek apakah frontend sedang mengecek transaksi if ( e && e.parameter && e.parameter.action === "check" ) { const clientId = e.parameter.clientId; if (!clientId) { return jsonResponse({ success: false, message: "Client ID tidak ditemukan." }); } return checkTransaction(clientId); } // GET biasa return jsonResponse({ success: true, message: "API Monitoring Keuangan aktif." }); } // ===================================================== // POST // ===================================================== function doPost(e) { try { if ( !e || !e.postData || !e.postData.contents ) { throw new Error( "Data POST tidak ditemukan." ); } const data = JSON.parse(e.postData.contents); const hasil = submitForm(data); return jsonResponse({ success: true, message: "Data berhasil disimpan.", idTransaksi: hasil.idTransaksi, clientId: hasil.clientId, jumlahNota: hasil.jumlahNota, jumlahBarang: hasil.jumlahBarang }); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } // ===================================================== // JSON RESPONSE // ===================================================== function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } // ===================================================== // DRIVE // ===================================================== function getFolder() { return DriveApp.getFolderById( FOLDER_ID ); } // ===================================================== // SHEET // ===================================================== function getSheet() { const ss = SpreadsheetApp.openById( SPREADSHEET_ID ); let sheet = ss.getSheetByName( NAMA_SHEET ); if (!sheet) { sheet = ss.insertSheet( NAMA_SHEET ); } setupHeader(sheet); return sheet; } // ===================================================== // SETUP HEADER // ===================================================== function setupHeader(sheet) { const lastColumn = sheet.getLastColumn(); // Sheet benar-benar kosong if (lastColumn === 0) { sheet .getRange( 1, 1, 1, HEADER_SHEET.length ) .setValues([ HEADER_SHEET ]) .setFontWeight("bold"); return; } const headerLama = sheet .getRange( 1, 1, 1, lastColumn ) .getValues()[0] .map(function(header) { return String(header).trim(); }); // Tambahkan header yang belum ada const headerKurang = HEADER_SHEET.filter( function(header) { return !headerLama.includes( header ); } ); if ( headerKurang.length === 0 ) { return; } const kolomMulai = sheet.getLastColumn() + 1; sheet .getRange( 1, kolomMulai, 1, headerKurang.length ) .setValues([ headerKurang ]) .setFontWeight("bold"); } // ===================================================== // SUBMIT FORM // ===================================================== function submitForm(data) { // ------------------------- // VALIDASI DASAR // ------------------------- if (!data) { throw new Error( "Data formulir kosong." ); } if (!data.nama) { throw new Error( "Nama panggilan belum diisi." ); } if (!data.divisi) { throw new Error( "Divisi belum dipilih." ); } if ( !data.notas || !Array.isArray(data.notas) ) { throw new Error( "Data nota tidak ditemukan." ); } if ( data.notas.length === 0 ) { throw new Error( "Minimal harus ada satu nota." ); } // ------------------------- // CLIENT ID // ------------------------- const clientId = data.clientId || Utilities.getUuid(); // ------------------------- // SHEET & DRIVE // ------------------------- const sheet = getSheet(); const folder = getFolder(); // ------------------------- // ID TRANSAKSI // ------------------------- const sekarang = new Date(); const timezone = Session.getScriptTimeZone(); const idTransaksi = "TRX-" + Utilities.formatDate( sekarang, timezone, "yyyyMMdd-HHmmss" ); const timestamp = sekarang; const rows = []; // =================================================== // PROSES NOTA // =================================================== data.notas.forEach( function(nota, indexNota) { let linkBukti = ""; // ------------------------- // UPLOAD FILE // ------------------------- if ( nota.file && nota.file.data ) { const bytes = Utilities.base64Decode( nota.file.data ); const blob = Utilities.newBlob( bytes, nota.file.mimeType, nota.file.name ); const file = folder.createFile( blob ); linkBukti = file.getUrl(); } // ------------------------- // VALIDASI BARANG // ------------------------- if ( !nota.barang || !Array.isArray( nota.barang ) || nota.barang.length === 0 ) { throw new Error( "Nota ke-" + (indexNota + 1) + " belum memiliki barang." ); } // ================================================= // PROSES BARANG // ================================================= nota.barang.forEach( function(item) { const jumlah = Number( item.jumlah ); const harga = Number( item.harga ); // ------------------------- // VALIDASI NAMA // ------------------------- if (!item.nama) { throw new Error( "Nama barang pada nota ke-" + (indexNota + 1) + " belum diisi." ); } // ------------------------- // VALIDASI JUMLAH // ------------------------- if ( isNaN(jumlah) || jumlah <= 0 ) { throw new Error( "Jumlah barang tidak valid." ); } // ------------------------- // VALIDASI HARGA // ------------------------- if ( isNaN(harga) || harga < 0 ) { throw new Error( "Harga barang tidak valid." ); } // ------------------------- // TOTAL // ------------------------- const total = jumlah * harga; // ------------------------- // ROW // ------------------------- rows.push([ idTransaksi, timestamp, data.nama, data.divisi, nota.tanggal, indexNota + 1, linkBukti, jumlah, item.nama, harga, total, clientId ]); } ); } ); // =================================================== // SIMPAN KE SHEET // =================================================== if ( rows.length === 0 ) { throw new Error( "Tidak ada data barang yang dapat disimpan." ); } /* * Pastikan header memiliki * 12 kolom. */ const startRow = sheet.getLastRow() + 1; sheet .getRange( startRow, 1, rows.length, HEADER_SHEET.length ) .setValues( rows ); // =================================================== // RETURN HASIL // =================================================== return { idTransaksi: idTransaksi, clientId: clientId, jumlahNota: data.notas.length, jumlahBarang: rows.length }; } // ===================================================== // CEK TRANSAKSI // ===================================================== function checkTransaction( clientId ) { try { const sheet = getSheet(); const data = sheet.getDataRange() .getValues(); if ( data.length <= 1 ) { return jsonResponse({ success: true, found: false, message: "Transaksi belum ditemukan." }); } /* * Client ID berada di * kolom ke-12. * * Array JavaScript dimulai * dari index 0. * * Jadi: * index 11 = Client ID */ for ( let i = 1; i < data.length; i++ ) { const clientIdSheet = String( data[i][11] ).trim(); if ( clientIdSheet === String(clientId).trim() ) { return jsonResponse({ success: true, found: true, message: "Transaksi ditemukan.", idTransaksi: data[i][0], clientId: data[i][11] }); } } return jsonResponse({ success: true, found: false, message: "Transaksi belum ditemukan." }); } catch (error) { return jsonResponse({ success: false, found: false, message: error.message }); } }
